

// funciones de orden superior

// 1. reduce
const precios = [120, 200, 350, 100, 125, 175];

// usamos una funcion para sumar (reducir) todos los precios del arreglo
// a una sola expresion
const total = precios.reduce((acumulador, currentNum) => acumulador + currentNum, 0);

console.log(total);

// 2. map, aplicamos una funcion dada a todos los elementos de una lista
const duplica = precios.map( (n) => n * 2); // el () no es necesario en n
console.log(duplica);
